import { useMemo } from 'react';

import type { FichasApi } from '../../../application/ports/FichasApi.js';
import { useFichas } from '../../../application/useFichas.js';
import styles from '../../../App.module.css';
import { HttpFichasApi } from '../../out/http/HttpFichasApi.js';

interface Props {
  api?: FichasApi;
  apiBaseUrl?: string;
  onViewAll?(): void;
}

export function ActiveFichasWidget({ api, apiBaseUrl = '/api/v1', onViewAll }: Props) {
  const resolvedApi = useMemo(() => api ?? new HttpFichasApi(apiBaseUrl), [api, apiBaseUrl]);
  const { pagination, isLoading, error, retry } = useFichas(resolvedApi, {
    activeOnly: true,
    page: 1,
    pageSize: 1,
  });

  return (
    <section className={styles.widget} aria-label="Fichas activas">
      <div className={styles.widgetTitle}>
        <svg aria-hidden="true" viewBox="0 0 16 16">
          <path d="M2 14V9m4 5V5m4 9V7m4 7V2" />
        </svg>
        <span>Fichas activas</span>
      </div>
      {isLoading && <strong aria-label="Cargando fichas activas">—</strong>}
      {error && <button type="button" onClick={retry}>Reintentar</button>}
      {!isLoading && !error && <strong>{pagination?.total_items ?? 0}</strong>}
      {onViewAll && <button type="button" className={styles.linkButton} onClick={onViewAll}>Ver todas</button>}
    </section>
  );
}

export default ActiveFichasWidget;

import { type FormEvent, useState } from 'react';

import {
  EMPTY_COMPETENCY_FORM,
  validateCompetencyForm,
  type CompetencyFormErrors,
  type CompetencyFormValues,
} from '../../../domain/trainingProgram.js';
import styles from './components.module.css';

interface Props {
  isSaving: boolean;
  error: Error | null;
  onSubmit(values: CompetencyFormValues): void;
  onClose(): void;
}

// E3: crear competencia dentro de un programa.
export function CompetencyFormModal({ isSaving, error, onSubmit, onClose }: Props) {
  const [values, setValues] = useState<CompetencyFormValues>(EMPTY_COMPETENCY_FORM);
  const [errors, setErrors] = useState<CompetencyFormErrors>({});

  const set = (key: keyof CompetencyFormValues, value: string) =>
    setValues((current) => ({ ...current, [key]: value }));

  const handleSubmit = (event: FormEvent) => {
    event.preventDefault();
    const found = validateCompetencyForm(values);
    setErrors(found);
    if (Object.keys(found).length === 0) onSubmit(values);
  };

  return (
    <div className={styles.backdrop} role="presentation" onMouseDown={onClose}>
      <div
        className={styles.dialog}
        role="dialog"
        aria-modal="true"
        aria-labelledby="competency-form-title"
        onMouseDown={(event) => event.stopPropagation()}
      >
        <h2 id="competency-form-title">Nueva competencia</h2>

        <form className={styles.form} onSubmit={handleSubmit} noValidate>
          <label className={styles.field}>
            <span>Código SENA</span>
            <input value={values.sena_code} onChange={(event) => set('sena_code', event.target.value)} />
            {errors.sena_code && <span className={styles.fieldError}>{errors.sena_code}</span>}
          </label>

          <label className={styles.field}>
            <span>Nombre</span>
            <input value={values.name} onChange={(event) => set('name', event.target.value)} />
            {errors.name && <span className={styles.fieldError}>{errors.name}</span>}
          </label>

          <label className={styles.field}>
            <span>Horas</span>
            <input inputMode="numeric" value={values.hours} onChange={(event) => set('hours', event.target.value)} />
            {errors.hours && <span className={styles.fieldError}>{errors.hours}</span>}
          </label>

          {error && <div className={styles.errorBox} role="alert">{error.message}</div>}

          <div className={styles.actions}>
            <button className={styles.secondaryButton} type="button" onClick={onClose}>Cancelar</button>
            <button className={styles.primaryButton} type="submit" disabled={isSaving}>
              {isSaving ? 'Guardando…' : 'Guardar competencia'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

import { useState } from 'react';

import type { TrainingProgramsApi } from '../../../application/ports/TrainingProgramsApi.js';
import { useLearningOutcomes } from '../../../application/useLearningOutcomes.js';
import { useProgramCompetencies } from '../../../application/useProgramCompetencies.js';
import { toCompetencyCreate, type TrainingProgram } from '../../../domain/trainingProgram.js';
import { CompetencyFormModal } from './CompetencyFormModal.js';
import styles from './components.module.css';
import { ProgramActiveBadge } from './ProgramActiveBadge.js';

interface OutcomesProps {
  api: TrainingProgramsApi;
  competencyId: string;
}

// E4: se monta solo al expandir la competencia, así los RAP se piden bajo demanda.
function LearningOutcomesList({ api, competencyId }: OutcomesProps) {
  const outcomes = useLearningOutcomes(api, competencyId);

  if (outcomes.isLoading) return <p className={styles.muted}>Cargando resultados de aprendizaje…</p>;

  if (outcomes.error) {
    return (
      <div className={styles.errorBox} role="alert">
        <p>{outcomes.error.message}</p>
        <button className={styles.secondaryButton} type="button" onClick={outcomes.retry}>Reintentar</button>
      </div>
    );
  }

  if (outcomes.data.length === 0) {
    return <p className={styles.muted}>Esta competencia no tiene resultados de aprendizaje.</p>;
  }

  return (
    <ul className={styles.outcomes} aria-label="Resultados de aprendizaje">
      {outcomes.data.map((outcome) => (
        <li key={outcome.id} className={styles.outcomeItem}>
          <strong>{outcome.code}</strong>
          <span>{outcome.description}</span>
          {!outcome.is_active && <ProgramActiveBadge isActive={false} />}
        </li>
      ))}
    </ul>
  );
}

interface Props {
  api: TrainingProgramsApi;
  program: TrainingProgram;
  canManage: boolean;
}

// E3 + E4: programa > competencias > RAP.
export function CurriculumTree({ api, program, canManage }: Props) {
  const competencies = useProgramCompetencies(api, program.id);
  const [expanded, setExpanded] = useState<ReadonlySet<string>>(new Set());
  const [isCreating, setIsCreating] = useState(false);

  const toggle = (id: string) =>
    setExpanded((current) => {
      const next = new Set(current);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });

  const openCreate = () => {
    competencies.clearActionError();
    setIsCreating(true);
  };

  const total = competencies.pagination?.total_items ?? competencies.data.length;

  return (
    <section className={styles.tree} aria-label="Currículo del programa">
      <div className={styles.treeHeader}>
        <div>
          <h2>Currículo de {program.name}</h2>
          <p className={styles.muted}>Versión vigente {program.version} · {total} competencias</p>
        </div>
        {canManage && (
          <button className={styles.primaryButton} type="button" disabled={!program.is_active} onClick={openCreate}>
            Nueva competencia
          </button>
        )}
      </div>

      {competencies.isLoading && <p className={styles.muted}>Cargando competencias…</p>}

      {competencies.error && (
        <div className={styles.errorBox} role="alert">
          <p>{competencies.error.message}</p>
          <button className={styles.secondaryButton} type="button" onClick={competencies.retry}>Reintentar</button>
        </div>
      )}

      {!competencies.isLoading && !competencies.error && competencies.data.length === 0 && (
        <p className={styles.muted}>Este programa aún no tiene competencias.</p>
      )}

      {competencies.data.length > 0 && (
        <ul className={styles.competencyList}>
          {competencies.data.map((competency) => {
            const isOpen = expanded.has(competency.id);
            const panelId = `outcomes-${competency.id}`;

            return (
              <li key={competency.id} className={styles.competencyItem}>
                <button
                  className={styles.competencyToggle}
                  type="button"
                  aria-expanded={isOpen}
                  aria-controls={panelId}
                  onClick={() => toggle(competency.id)}
                >
                  <span aria-hidden="true">{isOpen ? '▾' : '▸'}</span>
                  <span className={styles.code}>{competency.sena_code}</span>
                  <span className={styles.competencyName}>{competency.name}</span>
                  <span className={styles.muted}>{competency.hours} h</span>
                  {!competency.is_active && <ProgramActiveBadge isActive={false} />}
                </button>
                {isOpen && (
                  <div id={panelId} className={styles.outcomesPanel}>
                    <LearningOutcomesList api={api} competencyId={competency.id} />
                  </div>
                )}
              </li>
            );
          })}
        </ul>
      )}

      {competencies.pagination && competencies.pagination.total_pages > 1 && (
        <p className={styles.muted}>Se muestran las primeras {competencies.data.length} competencias.</p>
      )}

      {isCreating && (
        <CompetencyFormModal
          isSaving={competencies.isSaving}
          error={competencies.actionError}
          onClose={() => setIsCreating(false)}
          onSubmit={async (values) => {
            if (await competencies.create(toCompetencyCreate(values))) setIsCreating(false);
          }}
        />
      )}
    </section>
  );
}

import {
  EMPTY_PROGRAM_FORM,
  TRAINING_LEVELS,
  TRAINING_LEVEL_LABELS,
  programToFormValues,
  validateProgramForm,
  type KnowledgeNetwork,
  type ProgramFormErrors,
  type ProgramFormValues,
  type TrainingLevel,
  type TrainingProgram,
} from '../../../domain/trainingProgram.js';
import styles from './components.module.css';

interface Props {
  program?: TrainingProgram;
  networks: KnowledgeNetwork[];
  networksError: Error | null;
  isSaving: boolean;
  error: Error | null;
  onSubmit(values: ProgramFormValues): void;
  onClose(): void;

// E1: crear. E2: editar (subir la versión en 1 lo resuelve el dominio).
export function ProgramFormModal({ program, networks, networksError, isSaving, error, onSubmit, onClose }: Props) {
  const [values, setValues] = useState<ProgramFormValues>(() =>
    program ? programToFormValues(program) : EMPTY_PROGRAM_FORM,
  );
  const [errors, setErrors] = useState<ProgramFormErrors>({});
  const title = program ? 'Editar programa' : 'Nuevo programa';

  const set = <K extends keyof ProgramFormValues>(key: K, value: ProgramFormValues[K]) =>
    setValues((current) => ({ ...current, [key]: value }));

  const handleSubmit = (event: FormEvent) => {
    event.preventDefault();
    const found = validateProgramForm(values);
    setErrors(found);
    if (Object.keys(found).length === 0) onSubmit(values);
  };

  const hasCurrentNetwork = networks.some((network) => network.id === values.knowledge_network_id);

  return (
    <div className={styles.backdrop} role="presentation" onMouseDown={onClose}>
      <div
        className={styles.dialog}
        role="dialog"
        aria-modal="true"
        aria-labelledby="program-form-title"

        )}

        <form className={styles.form} onSubmit={handleSubmit} noValidate>
          <label className={styles.field}>
            <span>Red de conocimiento</span>

              onChange={(event) => set('knowledge_network_id', event.target.value)}
            >
              <option value="">Selecciona una red</option>
              {values.knowledge_network_id && !hasCurrentNetwork && (
                <option value={values.knowledge_network_id}>Red actual del programa</option>
              )}
              {networks.map((network) => (
                <option key={network.id} value={network.id}>{network.name}</option>
              ))}
            </select>
            {networksError && <span className={styles.fieldError}>No pudimos cargar las redes de conocimiento.</span>}
            {errors.knowledge_network_id && <span className={styles.fieldError}>{errors.knowledge_network_id}</span>}
          </label>

          <label className={styles.field}>
            <span>Código del programa</span>
            <input value={values.program_code} onChange={(event) => set('program_code', event.target.value)} />
            {errors.program_code && <span className={styles.fieldError}>{errors.program_code}</span>}
          </label>

          <label className={styles.field}>
            <span>Nombre</span>
            <input value={values.name} onChange={(event) => set('name', event.target.value)} />
            {errors.name && <span className={styles.fieldError}>{errors.name}</span>}
          </label>

          <label className={styles.field}>
            <span>Nivel de formación</span>
            <select
              value={values.training_level}
              onChange={(event) => set('training_level', event.target.value as TrainingLevel | '')}
            >
              <option value="">Selecciona un nivel</option>

                <option key={level} value={level}>{TRAINING_LEVEL_LABELS[level]}</option>
              ))}

            <span>Horas totales</span>

              inputMode="numeric"

              onChange={(event) => set('total_hours', event.target.value)}
            />
            {errors.total_hours && <span className={styles.fieldError}>{errors.total_hours}</span>}
          </label>

          {error && <div className={styles.errorBox} role="alert">{error.message}</div>}

          <div className={styles.actions}>

import type { FichasApi } from '../../../application/ports/FichasApi.js';
import type {
  ContractFicha,
  Ficha,
  FichasPageResult,
  ListFichasQuery,
} from '../../../domain/ficha.js';

interface ContractPage {
  data: Array<Omit<ContractFicha, 'status'> & { status: string }>;
  pagination: FichasPageResult['pagination'];
}

export class HttpFichasApi implements FichasApi {
  private readonly baseUrl: string;

  constructor(baseUrl = '/api/v1') {
    this.baseUrl = baseUrl.replace(/\/$/, '');
  }

  async list(query: ListFichasQuery, signal?: AbortSignal): Promise<FichasPageResult> {
    const params = new URLSearchParams();

    Object.entries(query).forEach(([key, value]) => {
      if (value !== undefined && value !== '') params.set(key, String(value));
    });

    const response = await fetch(`${this.baseUrl}/fichas?${params.toString()}`, {
      credentials: 'include',
      headers: { Accept: 'application/json' },
      signal,
    });

    return this.parseResponse<ContractPage>(response);
  }

  async getById(id: string, signal?: AbortSignal): Promise<Ficha> {
    const response = await fetch(`${this.baseUrl}/fichas/${encodeURIComponent(id)}`, {
      credentials: 'include',
      headers: { Accept: 'application/json' },
      signal,
    });

    return this.parseResponse<Ficha>(response);
  }

  private async parseResponse<T>(response: Response): Promise<T> {
    if (!response.ok) {
      throw new Error(`No fue posible consultar las fichas (HTTP ${response.status}).`);
    }

    return response.json() as Promise<T>;
  }
}

import type { TrainingProgramsApi } from '../../../application/ports/TrainingProgramsApi.js';
import type {
  Competency,
  CompetencyCreate,
  KnowledgeNetwork,
  LearningOutcome,
  ListCompetenciesQuery,
  ListKnowledgeNetworksQuery,
  ListLearningOutcomesQuery,
  ListTrainingProgramsQuery,
  Page,
  TrainingProgram,
  TrainingProgramCreate,
  TrainingProgramUpdate,
  TrainingProgramsPageResult,
} from '../../../domain/trainingProgram.js';

interface ErrorBody {
  error?: { message?: string };
}

const toQueryString = (query: object): string => {
  const params = new URLSearchParams();

  Object.entries(query).forEach(([key, value]) => {
    if (value !== undefined && value !== '') params.set(key, String(value));
  });

  const text = params.toString();
  return text ? `?${text}` : '';
};

export class HttpTrainingProgramsApi implements TrainingProgramsApi {
  private readonly baseUrl: string;

  constructor(baseUrl = '/api/v1') {
    this.baseUrl = baseUrl.replace(/\/$/, '');
  }

  list(query: ListTrainingProgramsQuery, signal?: AbortSignal): Promise<TrainingProgramsPageResult> {
    return this.request('GET', `/training-programs${toQueryString(query)}`, { signal });
  }

  create(body: TrainingProgramCreate): Promise<TrainingProgram> {
    return this.request('POST', '/training-programs', { body });
  }

  getById(id: string, signal?: AbortSignal): Promise<TrainingProgram> {
    return this.request('GET', `/training-programs/${encodeURIComponent(id)}`, { signal });
  }

  update(id: string, body: TrainingProgramUpdate): Promise<TrainingProgram> {
    return this.request('PUT', `/training-programs/${encodeURIComponent(id)}`, { body });
  }

  deactivate(id: string): Promise<void> {
    return this.request('DELETE', `/training-programs/${encodeURIComponent(id)}`);
  }

  listCompetencies(
    programId: string,
    query: ListCompetenciesQuery,
    signal?: AbortSignal,
  ): Promise<Page<Competency>> {
    return this.request(
      'GET',
      `/training-programs/${encodeURIComponent(programId)}/competencies${toQueryString(query)}`,
      { signal },
    );
  }

  createCompetency(programId: string, body: CompetencyCreate): Promise<Competency> {
    return this.request('POST', `/training-programs/${encodeURIComponent(programId)}/competencies`, { body });
  }

  listLearningOutcomes(
    competencyId: string,
    query: ListLearningOutcomesQuery,
    signal?: AbortSignal,
  ): Promise<Page<LearningOutcome>> {
    return this.request(
      'GET',
      `/competencies/${encodeURIComponent(competencyId)}/learning-outcomes${toQueryString(query)}`,
      { signal },
    );
  }

  listKnowledgeNetworks(
    query: ListKnowledgeNetworksQuery,
    signal?: AbortSignal,
  ): Promise<Page<KnowledgeNetwork>> {
    return this.request('GET', `/knowledge-networks${toQueryString(query)}`, { signal });
  }

  private async request<T>(
    method: 'GET' | 'POST' | 'PUT' | 'DELETE',
    path: string,
    options: { body?: unknown; signal?: AbortSignal } = {},
  ): Promise<T> {
    const headers: Record<string, string> = { Accept: 'application/json' };
    const hasBody = options.body !== undefined;
    if (hasBody) headers['Content-Type'] = 'application/json';

    const response = await fetch(`${this.baseUrl}${path}`, {
      method,
      credentials: 'include',
      headers,
      body: hasBody ? JSON.stringify(options.body) : undefined,
      signal: options.signal,
    });

    if (!response.ok) throw new Error(await this.errorMessage(response));
    if (response.status === 204) return undefined as T;

    return response.json() as Promise<T>;
  }

  // El contrato devuelve { error: { code, message } }; el mensaje ya viene en español.
  private async errorMessage(response: Response): Promise<string> {
    try {
      const body = (await response.json()) as ErrorBody;
      if (body.error?.message) return body.error.message;
    } catch {
      // cuerpo vacío o no JSON: se usa el mensaje genérico
    }

    return `No fue posible completar la operación (HTTP ${response.status}).`;
  }
}

  return { data, isLoading, error, retry };
}

import { useCallback, useEffect, useState } from 'react';

import type { TrainingProgramsApi } from './ports/TrainingProgramsApi.js';
import type { Competency, CompetencyCreate, ListCompetenciesQuery, Page } from '../domain/trainingProgram.js';

const COMPETENCIES_QUERY: ListCompetenciesQuery = { page: 1, page_size: 100 };
const NO_COMPETENCIES: Competency[] = [];

// E3: listar y crear competencias de un programa.
export function useProgramCompetencies(api: TrainingProgramsApi, programId?: string) {
  const [result, setResult] = useState<Page<Competency> | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const [reloadKey, setReloadKey] = useState(0);
  const [isSaving, setIsSaving] = useState(false);
  const [actionError, setActionError] = useState<Error | null>(null);

  useEffect(() => {
    if (!programId) {
      setResult(null);
      setError(null);
      setIsLoading(false);
      return;
    }

    const controller = new AbortController();
    setIsLoading(true);
    setError(null);

    void api
      .listCompetencies(programId, COMPETENCIES_QUERY, controller.signal)
      .then(setResult)
      .catch((cause: unknown) => {
        if (!controller.signal.aborted) {
          setError(cause instanceof Error ? cause : new Error('Error inesperado al consultar competencias.'));
        }
      })
      .finally(() => {
        if (!controller.signal.aborted) setIsLoading(false);
      });

    return () => controller.abort();
  }, [api, programId, reloadKey]);

  const retry = useCallback(() => setReloadKey((key) => key + 1), []);
  const clearActionError = useCallback(() => setActionError(null), []);

  const create = useCallback(
    async (body: CompetencyCreate): Promise<boolean> => {
      if (!programId) return false;
      setIsSaving(true);
      setActionError(null);
      try {
        await api.createCompetency(programId, body);
        setReloadKey((key) => key + 1);
        return true;
      } catch (cause) {
        setActionError(cause instanceof Error ? cause : new Error('Error inesperado al crear la competencia.'));
        return false;
      } finally {
        setIsSaving(false);
      }
    },
    [api, programId],
  );

  return {
    data: result?.data ?? NO_COMPETENCIES,
    pagination: result?.pagination,
    isLoading,
    error,
    retry,
    isSaving,
    actionError,
    clearActionError,
    create,
  };
}

  const clearActionError = useCallback(() => setActionError(null), []);

  const mutate = useCallback(async (action: () => Promise<unknown>): Promise<boolean> => {
    setIsSaving(true);
    setActionError(null);
    try {
      await action();
      setReloadKey((key) => key + 1);
      return true;
    } catch (cause) {
      setActionError(cause instanceof Error ? cause : new Error('Error inesperado al guardar.'));
      return false;
    } finally {
      setIsSaving(false);
    }
  }, []);

  const create = useCallback((body: TrainingProgramCreate) => mutate(() => api.create(body)), [api, mutate]);
  const update = useCallback(
    (id: string, body: TrainingProgramUpdate) => mutate(() => api.update(id, body)),
    [api, mutate],
  );
  const deactivate = useCallback((id: string) => mutate(() => api.deactivate(id)), [api, mutate]);

  return {
    data: result?.data ?? NO_PROGRAMS,
    pagination: result?.pagination,
    isLoading,
    error,
    retry,
    query,
    isSaving,
    actionError,
    clearActionError,
    create,
    update,
    deactivate,
  };
}

const isPositiveInteger = (raw: string) => /^\d+$/.test(raw.trim()) && Number(raw) > 0;

export function validateProgramForm(values: ProgramFormValues): ProgramFormErrors {
  const errors: ProgramFormErrors = {};
  if (!values.knowledge_network_id) errors.knowledge_network_id = 'Selecciona una red de conocimiento.';
  if (!values.program_code.trim()) errors.program_code = 'El código del programa es obligatorio.';
  if (!values.name.trim()) errors.name = 'El nombre es obligatorio.';
  if (!values.training_level) errors.training_level = 'Selecciona un nivel de formación.';
  if (!isPositiveInteger(values.total_hours)) errors.total_hours = 'Ingresa las horas como número entero mayor a 0.';
  return errors;
}

}

===== tests\unit\HttpTrainingProgramsApi.test.ts =====
import { afterEach, describe, expect, it, vi } from 'vitest';

import { HttpTrainingProgramsApi } from '../../src/adapter/out/http/HttpTrainingProgramsApi.js';
import { competenciesPage, outcomesPage, program, programsPage } from '../fixturesPrograms.js';

const json = (body: unknown, status = 200) =>
  new Response(JSON.stringify(body), { status, headers: { 'Content-Type': 'application/json' } });

const stubFetch = (response: Response) => {
  const fetchMock = vi.fn().mockResolvedValue(response);
  vi.stubGlobal('fetch', fetchMock);
  return fetchMock;
};

describe('HttpTrainingProgramsApi', () => {
  afterEach(() => vi.unstubAllGlobals());

  it('E1: traduce paginación y filtros a query params de GET /training-programs', async () => {
    const fetchMock = stubFetch(json(programsPage));
    const api = new HttpTrainingProgramsApi('/api/v1/');

    await api.list({ page: 2, page_size: 20, is_active: true, training_level: 'TECHNICIAN', program_code: '' });

    const url = String(fetchMock.mock.calls[0][0]);
    expect(url).toContain('/api/v1/training-programs?');
    expect(url).toContain('page=2');
    expect(url).toContain('page_size=20');
    expect(url).toContain('is_active=true');
    expect(url).toContain('training_level=TECHNICIAN');
    expect(url).not.toContain('program_code');
  });

  it('E1: crea con POST y cuerpo JSON', async () => {
    const fetchMock = stubFetch(json(program, 201));
    const api = new HttpTrainingProgramsApi();
    const body = {
      knowledge_network_id: program.knowledge_network_id,
      program_code: '1',
      name: 'N',
      training_level: 'OPERATOR' as const,
      total_hours: 10,
      is_active: true,
    };

    await expect(api.create(body)).resolves.toEqual(program);

    const [url, init] = fetchMock.mock.calls[0];
    expect(String(url)).toBe('/api/v1/training-programs');
    expect(init.method).toBe('POST');
    expect(init.headers['Content-Type']).toBe('application/json');
    expect(JSON.parse(init.body)).toEqual(body);
  });

  it('E2: actualiza con PUT /training-programs/{id}', async () => {
    const fetchMock = stubFetch(json({ ...program, version: 4 }));
    const api = new HttpTrainingProgramsApi();

    await api.update(program.id, { name: 'Otro', version: 4 });

    const [url, init] = fetchMock.mock.calls[0];
    expect(String(url)).toBe(`/api/v1/training-programs/${program.id}`);
    expect(init.method).toBe('PUT');
    expect(JSON.parse(init.body)).toEqual({ name: 'Otro', version: 4 });
  });

  it('E5: da de baja con DELETE y acepta 204 sin cuerpo', async () => {
    const fetchMock = stubFetch(new Response(null, { status: 204 }));
    const api = new HttpTrainingProgramsApi();

    await expect(api.deactivate(program.id)).resolves.toBeUndefined();

    const [url, init] = fetchMock.mock.calls[0];
    expect(String(url)).toBe(`/api/v1/training-programs/${program.id}`);
    expect(init.method).toBe('DELETE');
    expect(init.body).toBeUndefined();
  });

  it('E3: lista y crea competencias bajo el programa', async () => {
    const fetchMock = stubFetch(json(competenciesPage));
    const api = new HttpTrainingProgramsApi();

    await api.listCompetencies(program.id, { page: 1, page_size: 50 });
    expect(String(fetchMock.mock.calls[0][0])).toContain(
      `/api/v1/training-programs/${program.id}/competencies?page=1&page_size=50`,
    );

    fetchMock.mockResolvedValueOnce(json(competenciesPage.data[0], 201));
    await api.createCompetency(program.id, { name: 'C', sena_code: '1', hours: 10, is_active: true });
    expect(fetchMock.mock.calls[1][1].method).toBe('POST');
  });

  it('E4: lista los RAP de una competencia', async () => {
    const fetchMock = stubFetch(json(outcomesPage));
    const api = new HttpTrainingProgramsApi();

    await api.listLearningOutcomes('comp-1', { page: 1 });

    expect(String(fetchMock.mock.calls[0][0])).toBe('/api/v1/competencies/comp-1/learning-outcomes?page=1');
  });

  it('usa el mensaje del contrato cuando la API responde con error', async () => {
    stubFetch(json({ error: { code: 'CONFLICT', message: 'El código ya existe.' } }, 409));
    const api = new HttpTrainingProgramsApi();

    await expect(api.deactivate(program.id)).rejects.toThrow('El código ya existe.');
  });

  it('usa un mensaje genérico si el error no trae cuerpo JSON', async () => {
    stubFetch(new Response('boom', { status: 500 }));
    const api = new HttpTrainingProgramsApi();

    await expect(api.list({})).rejects.toThrow('HTTP 500');
  });
});

import { describe, expect, it } from 'vitest';

import {
  EMPTY_COMPETENCY_FORM,
  EMPTY_PROGRAM_FORM,
  canEditProgram,
  getActivePresentation,
  getTrainingLevelLabel,
  programToFormValues,
  toCompetencyCreate,
  toProgramCreate,
  toProgramUpdate,
  validateCompetencyForm,
  validateProgramForm,
} from '../../src/domain/trainingProgram.js';
import { inactiveProgram, program } from '../fixturesPrograms.js';

describe('trainingProgram (dominio)', () => {
  it('etiqueta los niveles y tolera uno desconocido', () => {
    expect(getTrainingLevelLabel('TECHNOLOGIST')).toBe('Tecnólogo');
    expect(getTrainingLevelLabel('NUEVO')).toBe('Nivel desconocido');
  });

  it('E5: presenta activo e inactivo', () => {
    expect(getActivePresentation(true).label).toBe('Activo');
    expect(getActivePresentation(false).label).toBe('Inactivo');
  });

  it('E2: solo permite editar programas activos', () => {
    expect(canEditProgram(program)).toBe(true);
    expect(canEditProgram(inactiveProgram)).toBe(false);
  });

  it('E1: valida el formulario vacío y uno completo', () => {
    expect(Object.keys(validateProgramForm(EMPTY_PROGRAM_FORM)).sort()).toEqual([
      'knowledge_network_id',
      'name',
      'program_code',
      'total_hours',
      'training_level',
    ]);
    expect(validateProgramForm(programToFormValues(program))).toEqual({});
  });

  it('E1: rechaza horas que no son enteros positivos', () => {
    const base = programToFormValues(program);
    expect(validateProgramForm({ ...base, total_hours: '0' }).total_hours).toBeDefined();
    expect(validateProgramForm({ ...base, total_hours: '12.5' }).total_hours).toBeDefined();
    expect(validateProgramForm({ ...base, total_hours: 'abc' }).total_hours).toBeDefined();
  });

  it('E1: arma el cuerpo de creación activo', () => {
    const body = toProgramCreate({ ...programToFormValues(program), name: '  Nuevo  ' });
    expect(body).toMatchObject({ name: 'Nuevo', total_hours: 2640, is_active: true });
  });

  it('E2: editar sube la versión en 1', () => {
    expect(toProgramUpdate(program, programToFormValues(program)).version).toBe(program.version + 1);
  });

  it('E3: valida y arma la competencia', () => {
    expect(Object.keys(validateCompetencyForm(EMPTY_COMPETENCY_FORM))).toHaveLength(3);
    expect(toCompetencyCreate({ sena_code: ' 1 ', name: 'X', hours: '40' })).toEqual({
      sena_code: '1',
      name: 'X',
      hours: 40,
      is_active: true,
    });
  });
});
